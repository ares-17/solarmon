```python
def solar_site():
    """Coordinate per alba/tramonto: prima LAT/LON esplicite, poi il capoluogo di PLACE."""
    if SOLAR_LAT and SOLAR_LON:
        try:
            lat, lon = float(SOLAR_LAT.replace(",", ".")), float(SOLAR_LON.replace(",", "."))
            if -90 <= lat <= 90 and -180 <= lon <= 180:
                return {"place": SOLAR_PLACE or "coordinate", "lat": lat, "lon": lon}
            log.warning("SOLAR_LAT/SOLAR_LON fuori range: ignorate")
        except ValueError:
            log.warning("SOLAR_LAT/SOLAR_LON non numeriche: ignorate")
    place = SOLAR_PLACE.strip()
    if not place:
        return None
    code = place.split(",")[-1].strip().upper()
    if code not in CAPITALS:
        code = next((k for k, v in CAPITALS.items() if v[0].casefold() == place.casefold()), "")
    hit = CAPITALS.get(code)
    if not hit:
        log.warning("SOLAR_PLACE non riconosciuto: %s", place)
        return None
    return {"place": place, "lat": hit[1], "lon": hit[2]}
```

```python
SUN_CACHE = {}


def sun_day(day, lat, lon):
    """Alba e tramonto (epoch UTC) per una data, algoritmo NOAA semplificato."""
    key = (day.isoformat(), round(lat, 3), round(lon, 3))
    if key in SUN_CACHE:
        return SUN_CACHE[key]
    n = day.timetuple().tm_yday
    g = 2 * math.pi / 365.0 * (n - 1 + 0.5)
    eq = 229.18 * (0.000075 + 0.001868 * math.cos(g) - 0.032077 * math.sin(g)
                   - 0.014615 * math.cos(2 * g) - 0.040849 * math.sin(2 * g))
    dec = (0.006918 - 0.399912 * math.cos(g) + 0.070257 * math.sin(g)
           - 0.006758 * math.cos(2 * g) + 0.000907 * math.sin(2 * g)
           - 0.002697 * math.cos(3 * g) + 0.00148 * math.sin(3 * g))
    cos_ha = (math.cos(math.radians(90.833)) / (math.cos(math.radians(lat)) * math.cos(dec))
              - math.tan(math.radians(lat)) * math.tan(dec))
    out = None
    if -1 <= cos_ha <= 1:
        ha = math.degrees(math.acos(cos_ha))
        base = calendar.timegm(day.timetuple())
        rise = base + round((720 - 4 * (lon + ha) - eq) * 60)
        sunset = base + round((720 - 4 * (lon - ha) - eq) * 60)
        if sunset > rise:
            out = (rise, sunset)
    if len(SUN_CACHE) > 800:
        SUN_CACHE.clear()
    SUN_CACHE[key] = out
    return out


def sun_payload(h):
    site = solar_site()
    if not site:
        return {"configured": False, "days": []}
    first = datetime.date(*time.gmtime(time.time() - h * 3600)[:3])
    last = datetime.date(*time.gmtime(time.time())[:3])
    days, day = [], first
    while day <= last:
        t = sun_day(day, site["lat"], site["lon"])
        if t:
            days.append({"date": day.isoformat(), "rise": t[0], "set": t[1]})
        day += datetime.timedelta(days=1)
    return {"configured": True, "place": site["place"], "lat": site["lat"],
            "lon": site["lon"], "days": days}
```

```python
def mix_payload(h):
    """Energia consumata nel periodo, divisa per fonte. Le ricariche non entrano."""
    since = time.time() - h * 3600
    with closing(connect()) as con:
        row = con.execute(
            "SELECT COALESCE(SUM(CASE WHEN mode<>'L' THEN MIN(COALESCE(pv_w,0), COALESCE(out_w,0)) ELSE 0 END),0),"
            " COALESCE(SUM(CASE WHEN mode<>'L' THEN MAX(0, COALESCE(out_w,0)-COALESCE(pv_w,0)) ELSE 0 END),0),"
            " COALESCE(SUM(CASE WHEN mode='L' THEN COALESCE(out_w,0) ELSE 0 END),0),"
            " COALESCE(SUM(COALESCE(out_w,0)),0), COUNT(*), MIN(ts), MAX(ts)"
            " FROM samples WHERE ts>?", (since,)).fetchone()
    solar, batt, grid, load, n, first, last = row
    k = STORE_INTERVAL / 3600000.0
    return {"h": h, "solar_kwh": round(solar * k, 3), "batt_kwh": round(batt * k, 3),
            "grid_kwh": round(grid * k, 3), "load_kwh": round(load * k, 3),
            "samples": n, "from": first, "to": last}
```

```markdown
## Mix del consumo e ore di luce (2026-11-13)

- Il mix attribuisce il **carico**, non il bilancio: FV diretto `min(pv_w, out_w)` fuori
  bypass, batteria `max(0, out_w - pv_w)` fuori bypass, rete tutto `out_w` quando
  `mode='L'`. Le ricariche restano fuori, così le tre voci chiudono sempre sul consumo.
- In bypass l'inverter passa l'ingresso AC al carico: attribuire tutto alla rete è la
  lettura prudente, anche se il FV sta caricando la batteria.
- `SOLAR_PLACE` risolve il **capoluogo** di provincia, non il comune: errore < ~10 km,
  irrilevante per alba/tramonto (1° di longitudine ≈ 4 minuti).
```

```js
const MIX_KEYS = [['solar', 'FV diretto'], ['batt', 'Batteria'], ['grid', 'Rete']];

function mixShares(vals) {
  const total = vals.reduce((a, b) => a + b, 0);
  const pct = total ? vals.map((v) => v / total * 100) : vals.map(() => 0);
  const out = pct.map(Math.floor);
  const left = total ? 100 - out.reduce((a, b) => a + b, 0) : 0;
  pct.map((v, i) => [v - Math.floor(v), i]).sort((a, b) => b[0] - a[0])
    .forEach(([, i], k) => { if (k < left) out[i]++; });
  return out;
}

function paintMix() {
  const meta = $('mix-meta'), ring = $('mix-ring'), key = $('mix-key'), bar = $('mix-bar');
  const d = MIX || {};
  const vals = [+d.solar_kwh || 0, +d.batt_kwh || 0, +d.grid_kwh || 0];
  const total = vals.reduce((a, b) => a + b, 0);
  ring.replaceChildren();
  key.replaceChildren();
  bar.replaceChildren();
  meta.textContent = total > 0 ? fmt(total, 1) + ' kWh consumati · ' + RANGE[H] : 'nessun consumo nel periodo';
  if (!total) return;
  const pct = mixShares(vals);
  const C = 2 * Math.PI * 50;
  const svg = el('svg', { viewBox: '0 0 132 132', width: 132, height: 132 }, ring);
  let acc = 0;
  vals.forEach((v, i) => {
    if (v <= 0) return;
    const len = v / total * C;
    el('circle', {
      cx: 66, cy: 66, r: 50, fill: 'none', stroke: `var(--mix-${MIX_KEYS[i][0]})`,
      'stroke-width': 14, 'stroke-dasharray': `${Math.max(0, len - 1.5).toFixed(2)} ${C.toFixed(2)}`,
      'stroke-dashoffset': (-acc).toFixed(2), transform: 'rotate(-90 66 66)'
    }, svg);
    acc += len;
  });
  txt(svg, 66, 63, fmt(total, 1), { 'text-anchor': 'middle', 'font-size': 17, 'font-weight': 600, fill: 'var(--text)' });
  txt(svg, 66, 79, 'kWh', { 'text-anchor': 'middle', 'font-size': 11, fill: 'var(--muted)' });
  vals.forEach((v, i) => {
    if (v <= 0) return;
    const li = document.createElement('li');
    li.innerHTML = `<i style="background:var(--mix-${MIX_KEYS[i][0]})"></i>` +
      `<span class="k">${MIX_KEYS[i][1]}</span><span class="p">${pct[i]}%</span>` +
      `<span class="v">${fmt(v, 2)} kWh</span>`;
    key.appendChild(li);
  });
  const stack = document.createElement('div');
  stack.className = 'mix-stack';
  vals.forEach((v, i) => {
    if (v <= 0) return;

    seg.style.width = (v / total * 100).toFixed(2) + '%';
    seg.style.background = `var(--mix-${MIX_KEYS[i][0]})`;
    seg.style.color = i === 0 ? 'var(--surface)' : 'var(--text)';
    if (pct[i] >= 12) seg.textContent = pct[i] + '%';
    stack.appendChild(seg);
  });
  bar.appendChild(stack);
  const ul = document.createElement('ul');
  ul.className = 'mix-inline';
  vals.forEach((v, i) => {
    if (v <= 0) return;
    const li = document.createElement('li');
    li.innerHTML = `<i style="background:var(--mix-${MIX_KEYS[i][0]})"></i>${MIX_KEYS[i][1]} ${pct[i]}%`;
    ul.appendChild(li);
  });
  bar.appendChild(ul);
}
```

```js
function timeIndex(ts, t) {
  if (t <= ts[0]) return 0;
  if (t >= ts[ts.length - 1]) return ts.length - 1;
  let lo = 0, hi = ts.length - 1;
  while (hi - lo > 1) {
    const mid = (lo + hi) >> 1;
    if (ts[mid] <= t) lo = mid; else hi = mid;
  }
  const span = ts[hi] - ts[lo] || 1;
  return lo + (t - ts[lo]) / span;
}
```
